func scalar(field int,value uint64)[]byte{return append(proto.EncodeVarint(uint64(field<<3)),proto.EncodeVarint(value)...)}
func encode(field int,body []byte)[]byte{data:=proto.EncodeVarint(uint64(field<<3|2));data=append(data,proto.EncodeVarint(uint64(len(body)))...);return append(data,body...)}
func group(field int,body []byte)[]byte{data:=proto.EncodeVarint(uint64(field<<3|3));data=append(data,body...);return append(data,proto.EncodeVarint(uint64(field<<3|4))...)}
func official(t *testing.T,wire []byte)[]byte{t.Helper();c:=exec.Command("protoc","-I.","--decode=compat.complex.Sample","complex.proto","extensions.proto");c.Stdin=bytes.NewReader(wire);text,err:=c.CombinedOutput();if err!=nil{t.Fatalf("official decode: %s %v",text,err)};c=exec.Command("protoc","-I.","--encode=compat.complex.Sample","complex.proto","extensions.proto");c.Stdin=bytes.NewReader(text);out,err:=c.CombinedOutput();if err!=nil{t.Fatalf("official encode: %s %v",out,err)};return out}
func canonicalUnknowns(t *testing.T,wire []byte)string{t.Helper();c:=exec.Command("protoc","-I.","--decode=compat.complex.Sample","complex.proto","extensions.proto");c.Stdin=bytes.NewReader(wire);text,err:=c.CombinedOutput();if err!=nil{t.Fatalf("decode %s %v",text,err)};lines:=strings.Split(string(text),"\n");sort.Strings(lines);return strings.Join(lines,"\n")}
func TestComplexRoundtrip(t *testing.T){
 child:=&pb.Child{X:proto.Int32(0),Y:proto.Int32(2)}
 value:=&pb.Sample{Enums:map[string]pb.Closed{"a":1},Checked:map[string]string{"雪":""},Raw:map[string]string{string([]byte{255}):string([]byte{254})},Children:map[int32]*pb.Child{1:child},ImportedChild:&pb.ForeignChild{},Prefixed:child,Delimited:child,Groups:[]*pb.Child{child,{}},Choice:&pb.Sample_SelectedGroup{SelectedGroup:child}}
 proto.SetExtension(value,pb.E_ForeignRaw,proto.String(string([]byte{255})));proto.SetExtension(value,pb.E_Declaring_NestedExt,proto.Int32(0));proto.SetExtension(value,pb.E_EnumExt,pb.Closed(1).Enum());proto.SetExtension(value,pb.E_EnumList,[]pb.Closed{1,2});proto.SetExtension(value,pb.E_GroupExt,child);proto.SetExtension(value,pb.E_RawExt,proto.String(string([]byte{255})))
 wire,err:=proto.Marshal(value);if err!=nil{t.Fatal(err)}
 if len(wire)!=proto.Size(value){t.Fatal("wrong size")}
 if FAST {if _,ok:=interface{}(value).(interface{Marshal()([]byte,error);Unmarshal([]byte)error;Size()int});!ok{t.Fatal("fast codec missing")}}
 var back pb.Sample;if err:=proto.Unmarshal(wire,&back);err!=nil||!proto.Equal(value,&back){t.Fatalf("roundtrip %v: %v",err,&back)}
 var reference pb.Sample;if err:=proto.Unmarshal(official(t,wire),&reference);err!=nil||!proto.Equal(value,&reference){t.Fatalf("oracle roundtrip %v: %v",err,&reference)}
}

 for _,field:=range []int{5,6,8,11}{
  
  wrap:=encode;if field==6||field==11{wrap=group}

  var value pb.Sample;if err:=proto.Unmarshal(data,&value);err!=nil{t.Fatal(err)}
  wire,_:=proto.Marshal(&value);if !bytes.Equal(official(t,data),official(t,wire)){t.Fatalf("message merge differs at %d: %x",field,wire)}
 }

func extendedTag(number,wire,extra int)[]byte{
 tag:=proto.EncodeVarint(uint64(number<<3|wire))
 if extra==0{return tag}
 tag[len(tag)-1]|=0x80
 for n:=1;n<extra;n++{tag=append(tag,0x80)}
 return append(tag,0)
}
func TestExtensionNonMinimalTags(t *testing.T){
 cases:=[]struct{name string;number,wire int;payload []byte;desc *proto.ExtensionDesc}{
  {"closed_enum",100,0,[]byte{1},pb.E_EnumExt},
  {"packed_enum",101,2,[]byte{2,1,2},pb.E_EnumList},
  {"delimited",102,3,scalar(1,1),pb.E_GroupExt},
  {"string",104,2,[]byte{1,'x'},pb.E_CheckedExt},
  {"nested_int32",108,0,[]byte{1},pb.E_Declaring_NestedExt},
  {"fixed32",109,5,[]byte{1,0,0,0},pb.E_Fixed32Ext},
  {"fixed64",110,1,[]byte{1,0,0,0,0,0,0,0},pb.E_Fixed64Ext},
  {"unregistered_middle",128,0,[]byte{1},nil},
  {"unregistered_last",199,0,[]byte{1},nil},
 }
 for _,c:=range cases{t.Run(c.name,func(t *testing.T){
  for extra:=0;extra<=3;extra++{for _,prefix:=range [][]byte{nil,encode(5,scalar(1,3))}{
   field:=append(extendedTag(c.number,c.wire,extra),c.payload...)
   if c.wire==3{field=append(field,extendedTag(c.number,4,extra)...)}
   input:=append(append(append([]byte{},prefix...),field...),scalar(250,11)...)
   var value pb.Sample
   if err:=proto.Unmarshal(input,&value);err!=nil{t.Fatalf("tag extra=%d offset=%d input=%x: %v",extra,len(prefix),input,err)}
   if len(prefix)>0&&(value.Prefixed==nil||value.Prefixed.GetX()!=3){t.Fatal("preceding field changed")}
   // Generated codecs preserve the original raw extension before lazy decoding.
   // The historical reflection codec may canonicalize the tag's varint bytes.
   wire,err:=proto.Marshal(&value);if err!=nil{t.Fatal(err)}
   if FAST&&!bytes.Equal(wire,input){t.Fatalf("raw extension changed: %x / %x",input,wire)}
   if proto.Size(&value)!=len(wire){t.Fatal("extension size differs")}
   expected:=canonicalUnknowns(t,input)
   if !FAST&&expected!=canonicalUnknowns(t,wire){t.Fatalf("official extension semantics differ: %x / %x",input,wire)}
   if c.desc!=nil{
    ext,err:=proto.GetExtension(&value,c.desc);if err!=nil{t.Fatalf("extension access: %v",err)}
    switch c.number{
    case 100: if got,ok:=ext.(*pb.Closed);!ok||*got!=1{t.Fatalf("enum extension=%v",ext)}
    case 101: if got,ok:=ext.([]pb.Closed);!ok||len(got)!=2||got[0]!=1||got[1]!=2{t.Fatalf("packed enum extension=%v",ext)}
    case 102: if got,ok:=ext.(*pb.Child);!ok||got.GetX()!=1{t.Fatalf("group extension=%v",ext)}
    case 104: if got,ok:=ext.(*string);!ok||*got!="x"{t.Fatalf("string extension=%v",ext)}
    case 108: if got,ok:=ext.(*int32);!ok||*got!=1{t.Fatalf("integer extension=%v",ext)}
    case 109: if got,ok:=ext.(*uint32);!ok||*got!=1{t.Fatalf("fixed32 extension=%v",ext)}
    case 110: if got,ok:=ext.(*uint64);!ok||*got!=1{t.Fatalf("fixed64 extension=%v",ext)}
    }
   }
  }}
 })}
}
