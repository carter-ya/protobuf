package qualify

import (
	"encoding/binary"
	"math"
	"reflect"
	"sort"
	"testing"

	pb "example.com/gogo-float-compare/pb"
	"github.com/gogo/protobuf/proto"
)

func varint(n uint64) []byte {
	var b []byte
	for n >= 128 {
		b = append(b, byte(n)|128)
		n >>= 7
	}
	return append(b, byte(n))
}
func field(n, wire int, data []byte) []byte {
	b := varint(uint64(n<<3 | wire))
	if wire == 2 {
		b = append(b, varint(uint64(len(data)))...)
	}
	return append(b, data...)
}
func single(n int) bool { return n == 1 || n == 3 || n == 5 || n == 7 || n == 11 }
func effective(n int, x float64) float64 {
	if single(n) {
		return float64(float32(x))
	}
	return x
}
func floatingWire(n int, values ...float64) []byte {
	wire := 1
	if single(n) {
		wire = 5
	}
	var payload []byte
	for i, value := range values {
		body := make([]byte, 8)
		if single(n) {
			body = body[:4]
			binary.LittleEndian.PutUint32(body, math.Float32bits(float32(value)))
		} else {
			binary.LittleEndian.PutUint64(body, math.Float64bits(value))
		}
		if n == 5 || n == 6 {
			entry := append(field(1, 2, []byte{byte('a' + i)}), field(2, wire, body)...)
			payload = append(payload, field(n, 2, entry)...)
		} else {
			payload = append(payload, body...)
		}
	}
	if n == 5 || n == 6 {
		return payload
	}
	if n == 3 || n == 4 {
		wire = 2
	}
	return field(n, wire, payload)
}
func decode(t *testing.T, data []byte) *pb.Record {
	t.Helper()
	got := new(pb.Record)
	if err := proto.Unmarshal(data, got); err != nil {
		t.Fatal(err)
	}
	return got
}
func TestFloatOrdering(t *testing.T) {
	values := []float64{math.Float64frombits(0x7ff8000000000001), math.Float64frombits(0xfff8000000000002), math.Inf(-1), -1, math.Copysign(0, -1), 0, 1, 1 + 1e-9, math.Inf(1)}
	for _, n := range []int{1, 2, 3, 4, 5, 6, 7, 8, 11, 12} {
		for _, a := range values {
			for _, b := range values {
				left, right := decode(t, floatingWire(n, a)), decode(t, floatingWire(n, b))
				order := sort.Float64Slice{effective(n, a), effective(n, b)}
				want := 0
				if order.Less(0, 1) {
					want = -1
				} else if order.Less(1, 0) {
					want = 1
				}
				if got := left.Compare(right); got != want {
					t.Fatalf("field %d: %v / %v = %d want %d", n, a, b, got, want)
				}
				if left.Compare(left) != 0 || right.Compare(left) != -want {
					t.Fatalf("field %d: non-reflexive or asymmetric order", n)
				}
			}
		}
		left := decode(t, append(floatingWire(n, math.NaN()), field(9, 0, []byte{1})...))
		right := decode(t, append(floatingWire(n, math.NaN()), field(9, 0, []byte{2})...))
		if left.Compare(right) >= 0 || right.Compare(left) <= 0 {
			t.Fatalf("field %d: NaN tie skipped next field", n)
		}
	}
	for _, n := range []int{3, 4, 5, 6} {
		left, right := decode(t, floatingWire(n, math.NaN(), 1)), decode(t, floatingWire(n, math.NaN(), 2))
		if left.Compare(right) >= 0 || right.Compare(left) <= 0 {
			t.Fatalf("field %d: NaN tie skipped next element", n)
		}
	}
	// Presence is checked only in layouts that represent it independently.
	for _, n := range []int{1, 2} {
		name := "ExplicitFloat"
		if n == 2 {
			name = "ExplicitDouble"
		}
		if reflect.ValueOf(new(pb.Record)).Elem().FieldByName(name).Kind() == reflect.Ptr {
			absent, present := decode(t, nil), decode(t, floatingWire(n, 0))
			if absent.Compare(present) >= 0 || present.Compare(absent) <= 0 {
				t.Fatal("pointer presence order changed")
			}
		}
	}
	for _, n := range []int{7, 8} {
		absent, present := decode(t, nil), decode(t, floatingWire(n, 0))
		if absent.Compare(present) >= 0 || present.Compare(absent) <= 0 {
			t.Fatal("oneof presence order changed")
		}
	}
}
