func same(a, b proto.Message) bool { return apiequality.Semantic.DeepEqual(a, b) }

func marshalJSON(value proto.Message) (string, error) {
	raw, err := json.Marshal(value)
	return string(raw), err
}
func unmarshalJSON(text string, value proto.Message) error {
	return json.Unmarshal([]byte(text), value)
}
func TestMaintainedUIDEquality(t *testing.T) {
	if os.Getenv("PROJECT_COMPAT_RUNTIME") != "fork" {
		defer func() {
			if r := recover(); r == nil || !strings.Contains(fmt.Sprint(r), "types.UID") {
				t.Fatalf("original UID panic control changed: %v", r)
			}
		}()
		proto.Equal(&metav1.ObjectMeta{}, &metav1.ObjectMeta{})
		return
	}
	a, b := &metav1.ObjectMeta{UID: types.UID("id")}, &metav1.ObjectMeta{UID: types.UID("id")}
	if !proto.Equal(a, b) {
		t.Fatal("equal real UID metadata differs")
	}
	b.UID = "other"
	if proto.Equal(a, b) {
		t.Fatal("different real UID metadata compares equal")
	}
}

func checkHelpers(t *testing.T, value proto.Message, check func()) {
	t.Helper()
	if os.Getenv("PROJECT_COMPAT_RUNTIME") == "original" && os.Getenv("PROJECT_COMPAT_VARIANT") == "purego" {
		defer func() {
			if r := recover(); r == nil || !strings.Contains(fmt.Sprint(r), "types.UID") {
				t.Fatalf("original purego clone panic control changed: %v", r)
			}
		}()
		proto.Clone(value)
		return
	}
	check()
}
