package projects

import (
	"context"
	stderrors "errors"
	errors "github.com/cockroachdb/errors"
	"github.com/cockroachdb/errors/extgrpc"
	"github.com/gogo/protobuf/proto"
	"google.golang.org/grpc/codes"
	"reflect"
	"testing"
)

func samples() []proto.Message {
	value := errors.EncodeError(context.Background(), errors.WithDetail(extgrpc.WrapWithGrpcCode(stderrors.New("missing"), codes.NotFound), "detail"))
	return []proto.Message{&value}
}
func TestErrorDetailsAndGrpc(t *testing.T) {
	original := errors.WithDetail(extgrpc.WrapWithGrpcCode(errors.New("missing"), codes.NotFound), "detail")
	encoded := errors.EncodeError(context.Background(), original)
	wire, err := proto.Marshal(&encoded)
	if err != nil {
		t.Fatal(err)
	}
	encoded.Reset()
	if err := proto.Unmarshal(wire, &encoded); err != nil {
		t.Fatal(err)
	}
	decoded := errors.DecodeError(context.Background(), encoded)
	if extgrpc.GetGrpcCode(decoded) != codes.NotFound || !reflect.DeepEqual(errors.GetAllDetails(decoded), []string{"detail"}) {
		t.Fatalf("grpc/details changed: %v", decoded)
	}
}

func same(a, b proto.Message) bool { return proto.Equal(a, b) }

func checkHelpers(t *testing.T, value proto.Message, check func()) { t.Helper(); check() }
