package consumer

import (
	pb "example.com/oneof-merge/pb"
	"github.com/gogo/protobuf/proto"
	"testing"
)

func TestOneofMessageMerging(t *testing.T) {
	for _, decode := range []func([]byte, *pb.Root) error{
		func(b []byte, m *pb.Root) error { return proto.Unmarshal(b, m) },
		func(b []byte, m *pb.Root) error { return proto.NewBuffer(b).Unmarshal(m) },
		func(b []byte, m *pb.Root) error { var info proto.InternalMessageInfo; return info.Unmarshal(m, b) },
	} {
		wires := [][]byte{
			{10, 2, 8, 1, 10, 2, 16, 2},
			{10, 2, 8, 1, 24, 7, 10, 2, 16, 2},
		}
		if MERGE {
			wires = append(wires, []byte{10, 2, 8, 1, 8, 7, 10, 2, 16, 2})
		}
		for _, wire := range wires {
			var got pb.Root
			if err := decode(wire, &got); err != nil {
				t.Fatal(err)
			}
			wantA := int32(0)
			if MERGE {
				wantA = 1
			}
			if got.GetChild() == nil || got.GetChild().GetA() != wantA || got.GetChild().GetB() != 2 {
				t.Fatalf("oneof message merge: %v", &got)
			}
		}
		var got pb.Root
		if err := decode([]byte{10, 2, 8, 1, 18, 0, 10, 2, 16, 2}, &got); err != nil || got.GetChild() == nil || got.GetChild().GetA() != 0 || got.GetChild().GetB() != 2 {
			t.Fatalf("oneof switch retained old message: %v %v", &got, err)
		}
	}
	if FAST {
		got := new(pb.Root)
		err := interface{}(got).(interface{ Unmarshal([]byte) error }).Unmarshal([]byte{10, 2, 8, 1, 10, 2, 16, 2})
		wantA := int32(0)
		if MERGE {
			wantA = 1
		}
		if err != nil || got.GetChild() == nil || got.GetChild().GetA() != wantA || got.GetChild().GetB() != 2 {
			t.Fatalf("direct oneof merge: %v %v", got, err)
		}
	}
	var message pb.Root
	if err := proto.Unmarshal([]byte{10, 2, 8, 1}, &message); err != nil {
		t.Fatal(err)
	}
	if err := proto.UnmarshalMerge([]byte{10, 2, 16, 2}, &message); err != nil {
		t.Fatal(err)
	}
	wantA := int32(0)
	if MERGE {
		wantA = 1
	}
	if message.GetChild() == nil || message.GetChild().GetA() != wantA {
		t.Fatalf("UnmarshalMerge contract: %v", &message)
	}
	message.Choice = (*pb.Root_Child)(nil)
	if err := proto.UnmarshalMerge([]byte{10, 0}, &message); err != nil || message.GetChild() == nil {
		t.Fatalf("typed nil wrapper: %v", err)
	}
}
