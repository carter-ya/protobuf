type oracle struct {
	command *exec.Cmd
	input   io.WriteCloser
	output  io.ReadCloser
	lock    sync.Mutex
}

var oracles [2]*oracle
var schemas [2]protoreflect.MessageDescriptor
var resolver *dynamicpb.Types

func TestMain(m *testing.M) {
	descriptorBytes, err := os.ReadFile(os.Getenv("GOGO_WIRE_DESCRIPTOR"))
	if err != nil {
		panic(err)
	}
	var set descriptorpb.FileDescriptorSet
	if err := modern.Unmarshal(descriptorBytes, &set); err != nil {
		panic(err)
	}
	files, err := protodesc.NewFiles(&set)
	if err != nil {
		panic(err)
	}
	resolver = dynamicpb.NewTypes(files)
	for i, name := range []string{"protobuf_test_messages.editions.TestAllTypesEdition2023", "qualification.closed.Record"} {
		desc, err := files.FindDescriptorByName(protoreflect.FullName(name))
		if err != nil {
			panic(err)
		}
		schemas[i] = desc.(protoreflect.MessageDescriptor)
		cmd := exec.Command(os.Getenv("PROTOBUF_CONFORMANCE_RUNNER"), "wire_oracle", name, os.Getenv("GOGO_WIRE_DESCRIPTOR"))
		// Invalid UTF-8 diagnostics from the C++ parser are expected during fuzzing.
		// A closed pipe or nonzero oracle exit is reported as a test failure.
		input, err := cmd.StdinPipe()
		if err != nil {
			panic(err)
		}
		output, err := cmd.StdoutPipe()
		if err != nil {
			panic(err)
		}
		if err := cmd.Start(); err != nil {
			panic(err)
		}
		oracles[i] = &oracle{command: cmd, input: input, output: output}
	}
	status := m.Run()
	for _, oracle := range oracles {
		oracle.input.Close()
		if err := oracle.command.Wait(); err != nil {
			fmt.Fprintln(os.Stderr, "binary oracle:", err)
			status = 1
		}
		oracle.output.Close()
	}
	os.Exit(status)
}
func (o *oracle) request(t *testing.T, frame []byte) bool {
	t.Helper()
	o.lock.Lock()
	defer o.lock.Unlock()
	header := make([]byte, 4)
	binary.LittleEndian.PutUint32(header, uint32(len(frame)))
	if _, err := o.input.Write(append(header, frame...)); err != nil {
		t.Fatal("oracle write:", err)
	}
	var response [1]byte
	if _, err := io.ReadFull(o.output, response[:]); err != nil {
		t.Fatal("oracle read:", err)
	}
	if response[0] > 1 {
		t.Fatalf("invalid oracle response %x", response)
	}
	return response[0] == 1
}
func message(which byte) proto.Message {
	if which&1 != 0 {
		return new(closed.Record)
	}
	return new(official.TestAllTypesEdition2023)
}
func varint(value uint64) []byte {
	var out [10]byte
	n := binary.PutUvarint(out[:], value)
	return out[:n]
}
func field(number int, wire int, value []byte) []byte {
	out := varint(uint64(number<<3 | wire))
	if wire == proto.WireBytes {
		out = append(out, varint(uint64(len(value)))...)
	}
	return append(out, value...)
}

func checkWire(t *testing.T, which byte, input []byte) {
	t.Helper()
	if len(input) > 4096 {
		t.Skip()
	}
	oracle := oracles[which&1]
	accepted := oracle.request(t, append([]byte{0}, input...))
	got := message(which)
	err := proto.Unmarshal(input, got)
	if _, envelopeError := strictEnvelope(input, schemas[which&1], 0); envelopeError != nil {
		if err == nil {
			t.Fatalf("SDK accepted invalid wire envelope %x: %v", input, envelopeError)
		}
		return
	}
	if !accepted {
		if err == nil {
			official := dynamicpb.NewMessage(schemas[which&1])
			if (modern.UnmarshalOptions{Resolver: resolver}).Unmarshal(input, official) != nil {
				t.Fatalf("SDK accepted wire rejected by C++ and Go %x", input)
			}
		}
		return
	}
	if err != nil {
		// C++ accepts overflowing tenth varint bytes that official Go rejects.
		// Require both independent parsers to accept before demanding SDK acceptance;
		// C++ alone remains the semantic oracle (Go dynamic has closed-enum gaps).
		official := dynamicpb.NewMessage(schemas[which&1])
		if (modern.UnmarshalOptions{Resolver: resolver}).Unmarshal(input, official) != nil {
			return
		}
		t.Fatalf("SDK rejected wire accepted by C++ and Go %x: %v", input, err)
	}
	output, err := proto.Marshal(got)
	if err != nil {
		t.Fatalf("SDK re-encode %x: %v", input, err)
	}
	compare := make([]byte, 5, 5+len(input)+len(output))
	compare[0] = 1
	binary.LittleEndian.PutUint32(compare[1:], uint32(len(input)))
	compare = append(compare, input...)
	compare = append(compare, output...)
	if !oracle.request(t, compare) {
		t.Fatalf("wire semantics changed: input %x / SDK %x", input, output)
	}
	if proto.Size(got) != len(output) {
		t.Fatalf("size %d differs from encoded length %d", proto.Size(got), len(output))
	}
}
func TestWireSeeds(t *testing.T) {
	for which := byte(0); which < 2; which++ {
		for i, input := range seeds() {
			t.Run(fmt.Sprintf("%d/%d", which, i), func(t *testing.T) { checkWire(t, which, input) })
		}
	}
}
func FuzzWire(f *testing.F) {
	for which := byte(0); which < 2; which++ {
		for _, input := range seeds() {
			f.Add(which, input)
		}
	}
	f.Fuzz(checkWire)
}

func TestGeneratedBooleanMapComparison(t *testing.T) {
	left := &official.TestAllTypesEdition2023{MapBoolBool: map[bool]bool{false: false}}
	right := &official.TestAllTypesEdition2023{MapBoolBool: map[bool]bool{false: true}}
	type comparer interface{ Compare(interface{}) int }
	compare, ok := interface{}(left).(comparer)
	if !ok {
		return
	} // Only auxiliary-enabled generators emit Compare.
	if compare.Compare(right) != -1 || interface{}(right).(comparer).Compare(left) != 1 || compare.Compare(left) != 0 {
		t.Fatal("boolean map comparison must order false before true")
	}
}

func TestGeneratedMapComparisonKeysAndOrder(t *testing.T) {
	type comparer interface{ Compare(interface{}) int }
	if _, ok := interface{}(new(official.TestAllTypesEdition2023)).(comparer); !ok {
		return
	}
	for _, pair := range [][2]*official.TestAllTypesEdition2023{
		{{MapBoolBool: map[bool]bool{false: false}}, {MapBoolBool: map[bool]bool{true: false}}},
		{{MapInt32Int32: map[int32]int32{1: 0}}, {MapInt32Int32: map[int32]int32{2: 0}}},
		{{MapStringString: map[string]string{"a": ""}}, {MapStringString: map[string]string{"b": ""}}},
		{{MapStringBytes: map[string][]byte{"a": nil}}, {MapStringBytes: map[string][]byte{"b": nil}}},
		{{MapStringNestedMessage: map[string]*official.TestAllTypesEdition2023_NestedMessage{"a": nil}}, {MapStringNestedMessage: map[string]*official.TestAllTypesEdition2023_NestedMessage{"b": nil}}},
		{{MapInt32Int32: map[int32]int32{1: 1, 2: 2}}, {MapInt32Int32: map[int32]int32{1: 2, 2: 1}}},
	} {
		a, b := interface{}(pair[0]).(comparer), interface{}(pair[1]).(comparer)
		for i := 0; i < 128; i++ {
			if a.Compare(pair[1]) != -1 || b.Compare(pair[0]) != 1 {
				t.Fatal("map comparison must distinguish keys, be antisymmetric and deterministic")
			}
		}
	}
}

func TestNativeEditionMessageExtensionValidation(t *testing.T) {
	for _, input := range [][]byte{
		field(100, proto.WireBytes, []byte{8}), field(100, proto.WireBytes, []byte{0}),
		field(100, proto.WireBytes, []byte{26, 1, 255}),
		field(104, proto.WireBytes, nil), field(104, proto.WireBytes, []byte{8, 1}),
		field(105, proto.WireBytes, []byte{8, 1}),
	} {
		if oracles[1].request(t, append([]byte{0}, input...)) {
			t.Fatalf("oracle accepted invalid extension %x", input)
		}
		if err := proto.Unmarshal(input, new(closed.Record)); err == nil {
			t.Fatalf("SDK accepted invalid registered message extension %x", input)
		}
	}
	input := append(field(104, proto.WireBytes, []byte{8, 1}), field(104, proto.WireBytes, []byte{16, 2})...)
	var got closed.Record
	if err := proto.Unmarshal(input, &got); err != nil {
		t.Fatalf("required fragments must merge before validation: %v", err)
	}
	value, err := proto.GetExtension(&got, closed.E_ExtensionRequired)
	if err != nil || value.(*closed.RequiredPayload).GetA() != 1 || value.(*closed.RequiredPayload).GetB() != 2 {
		t.Fatalf("required extension merge: %v", err)
	}
	checkWire(t, 1, input)
}
func TestDecodedEditionExtensionMerge(t *testing.T) {
	first, second := field(100, proto.WireBytes, []byte{8, 1}), field(100, proto.WireBytes, []byte{16, 2})
	for _, set := range []bool{false, true} {
		var got closed.Record
		if set {
			if err := proto.SetExtension(&got, closed.E_ExtensionPayload, &closed.Payload{A: proto.Int32(1)}); err != nil {
				t.Fatal(err)
			}
		} else {
			if err := proto.Unmarshal(first, &got); err != nil {
				t.Fatal(err)
			}
			if _, err := proto.GetExtension(&got, closed.E_ExtensionPayload); err != nil {
				t.Fatal(err)
			}
		}
		if err := proto.UnmarshalMerge(second, &got); err != nil {
			t.Fatal(err)
		}
		value, err := proto.GetExtension(&got, closed.E_ExtensionPayload)
		if err != nil || value.(*closed.Payload).GetA() != 1 || value.(*closed.Payload).GetB() != 2 {
			t.Fatalf("decoded extension lost incoming fields: %v %v", value, err)
		}
	}
}

func TestNestedRequiredExtensionFragments(t *testing.T) {
	input := append(field(110, proto.WireBytes, []byte{10, 2, 8, 1}), field(110, proto.WireBytes, []byte{10, 2, 16, 2})...)
	var got closed.Record
	if err := proto.Unmarshal(input, &got); err != nil {
		t.Fatal("nested required fragments must merge before validation", err)
	}
	value, err := proto.GetExtension(&got, closed.E_ExtensionNestedRequired)
	if err != nil || value.(*closed.RequiredContainer).GetChild().GetA() != 1 || value.(*closed.RequiredContainer).GetChild().GetB() != 2 {
		t.Fatalf("nested required extension: %v", err)
	}
	checkWire(t, 1, input)
}

func TestEditionExtensionFragmentValidation(t *testing.T) {
	cases := [][]byte{
		append(field(100, proto.WireBytes, []byte{26, 3, 'A'}), field(100, proto.WireBytes, []byte{8, 1})...),
		append(field(110, proto.WireBytes, []byte{10, 4, 8, 1}), field(110, proto.WireBytes, []byte{16, 2})...),
		field(110, proto.WireBytes, []byte{10, 2, 8, 1, 18, 1, 255, 10, 2, 16, 2, 18, 1, 'x'}),
	}
	entry := []byte{18, 2, 8, 1, 10, 1, 255, 18, 2, 16, 2, 10, 1, 'x'}
	body := append([]byte{10, 2, 8, 1}, field(3, proto.WireBytes, entry)...)
	body = append(body, 10, 2, 16, 2)
	cases = append(cases, field(110, proto.WireBytes, body))
	cases = append(cases, field(110, proto.WireBytes, field(3, proto.WireBytes, []byte{10, 1, 'x'})))
	for _, input := range cases {
		if oracles[1].request(t, append([]byte{0}, input...)) {
			t.Fatalf("oracle accepted invalid fragments %x", input)
		}
		var got closed.Record
		if err := proto.Unmarshal(input, &got); err == nil {
			t.Fatalf("SDK accepted invalid fragments %x", input)
		}
		number, extension := int32(100), closed.E_ExtensionPayload
		if input[0] != field(100, proto.WireBytes, nil)[0] {
			number, extension = 110, closed.E_ExtensionNestedRequired
		}
		proto.SetRawExtension(&got, number, input)
		if _, err := proto.GetExtension(&got, extension); err == nil {
			t.Fatalf("GetExtension accepted invalid fragments %x", input)
		}
	}
}

// Reference parsers may ignore overflowing tag bits while skipping an unknown
// group. Enforce the published 29-bit field-number range before comparing their
// acceptance. Unknown length-delimited payloads stay opaque.
func strictEnvelope(data []byte, schema protoreflect.MessageDescriptor, group protowire.Number) (int, error) {
	offset := 0
	for offset < len(data) {
		number, wire, header := protowire.ConsumeTag(data[offset:])
		if header < 0 || number < 1 || number > protowire.MaxValidNumber {
			return 0, fmt.Errorf("invalid field number")
		}
		offset += header
		if wire == protowire.EndGroupType {
			if group == 0 || number != group {
				return 0, fmt.Errorf("mismatched group")
			}
			return offset, nil
		}
		var field protoreflect.FieldDescriptor
		if schema != nil {
			field = schema.Fields().ByNumber(number)
			if field == nil {
				if extension, err := resolver.FindExtensionByNumber(schema.FullName(), number); err == nil {
					field = extension.TypeDescriptor()
				}
			}
		}
		if wire == protowire.StartGroupType {
			var child protoreflect.MessageDescriptor
			if field != nil && field.Kind() == protoreflect.GroupKind {
				child = field.Message()
			}
			consumed, err := strictEnvelope(data[offset:], child, number)
			if err != nil {
				return 0, err
			}
			offset += consumed
			continue
		}
		if wire == protowire.BytesType {
			body, consumed := protowire.ConsumeBytes(data[offset:])
			if consumed < 0 {
				return 0, fmt.Errorf("truncated bytes")
			}
			if field != nil && (field.Kind() == protoreflect.MessageKind || field.IsMap()) {
				if _, err := strictEnvelope(body, field.Message(), 0); err != nil {
					return 0, err
				}
			}
			offset += consumed
			continue
		}
		consumed := protowire.ConsumeFieldValue(number, wire, data[offset:])
		if consumed < 0 {
			return 0, fmt.Errorf("invalid field payload")
		}
		offset += consumed
	}
	if group != 0 {
		return 0, fmt.Errorf("unterminated group")
	}
	return offset, nil
}
