package types

import (
	"bytes"
	"encoding/json"
	"fmt"
)

// Private mutable state makes accidental reflection-based cloning observable.
type Raw struct{ data []byte }

func NewRaw(data []byte) *Raw                    { return &Raw{data: append([]byte{}, data...)} }
func (r Raw) Marshal() ([]byte, error)           { return append([]byte{}, r.data...), nil }
func (r Raw) Size() int                          { return len(r.data) }
func (r Raw) MarshalTo(data []byte) (int, error) { return copy(data, r.data), nil }
func (r *Raw) Unmarshal(data []byte) error       { r.data = append([]byte{}, data...); return nil }
func (r Raw) MarshalJSON() ([]byte, error)       { return json.Marshal(append([]byte{}, r.data...)) }
func (r *Raw) UnmarshalJSON(data []byte) error {
	var value []byte
	if err := json.Unmarshal(data, &value); err != nil {
		return err
	}
	return r.Unmarshal(value)
}
func (r Raw) Equal(other Raw) bool  { return bytes.Equal(r.data, other.data) }
func (r Raw) Compare(other Raw) int { return bytes.Compare(r.data, other.data) }
func (r Raw) GoString() string      { return fmt.Sprintf("types.NewRaw(%#v)", r.data) }
func (r *Raw) Mutate()              { r.data[0] ^= 255 }
func NewPopulatedRaw(r interface{ Intn(int) int }) *Raw {
	data := make([]byte, r.Intn(8))
	for i := range data {
		data[i] = byte(r.Intn(256))
	}
	return NewRaw(data)
}
